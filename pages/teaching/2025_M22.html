def moyenne (series):
return np.array(series).mean()

def moyenne_perso (series):
sum = 0
for i in range(0, len(series), 1):
sum += series[i]
return sum / len(series)

def variance (series):
return np.array(series).var()

def variance_perso (series):
moy = moyenne_perso(series)
sum = 0
for i in range(0, len(series), 1):
sum += (series[i] - moy) * (series[i] - moy)
return sum / len(series)

def ecart_type (series):
return np.array(series).std()

def vals_norm (moyenne, ecart_type, n):
rng = np.random.default_rng()
return rng.normal(loc = moyenne, scale = ecart_type, size = n)

def vals_unif (min_val, max_val, n):
rng = np.random.default_rng()
return rng.uniform(low = min_val, high = max_val, size = n)
