import numpy as np

def moyenne(series):
return np.array(series).mean()

def moyenne_perso(series):
sum = 0
for i in range(0, len(series), 1):
sum += series[i]
return sum / len(series)

def variance(series):
return np.array(series).var()

def variance_perso(series):
moy = moyenne_perso(series)
sum = 0
for i in range(0, len(series), 1):
sum += (series[i] - moy) * (series[i] - moy)
return sum / len(series)

x = np.arange(1, len(df) + 1, 1)

# régression linéaire pour norm
x_norm_bar = x.mean()
y_norm_bar = df["norm"].mean()
S_norm_xy = ((x - x_norm_bar) * (df["norm"] - y_norm_bar)).sum() / len(df)
S_norm_x2 = x.var()
a_norm = S_norm_xy / S_norm_x2
b_norm = y_norm_bar - a_norm * x_norm_bar
print("coefficients de régression linaire pour norm : ", a_norm, b_norm)

# régression linéaire pour unif
x_unif_bar = x.mean()
y_unif_bar = df["unif"].mean()
S_unif_xy = ((x - x_unif_bar) * (df["unif"] - y_unif_bar)).sum() / len(df)
S_unif_x2 = x.var()
a_unif = S_unif_xy / S_unif_x2
b_unif = y_unif_bar - a_unif * x_unif_bar
print("coefficients de régression linaire pour unif : ", a_unif, b_unif)

# tracé des droites de régression linéaire
