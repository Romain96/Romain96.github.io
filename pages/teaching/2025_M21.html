#define SIZE_SIGN_CODE 1
#define SIZE_EXP_CODE 8
#define SIZE_MANTISSE_CODE 23

union floatint
{
uint32_t i;// entier non signé sur 32 bits
float f;// flottant sur 32 bits
}

void i_code (uint32_t nb, uint8_t *bits)
{
for (size_t i = 0; i < 32; i++)
{
uint32_t* bit_i = (1 << (31 - i));// masque avec le ième bit à 1 et les autres à 0
bits[i] = (nb & bit_i) ? 1 : 0;// extraction du ième bit de nb
}
}

void u_code (union floatint fi, uint8_t *bits)
{
i_code(fi.i, bits);
}

void f_code (float f, uint8_t *bits)
{
union floatint fi;
fi.f = f;
u_code(fi, bits);
}

void get_info(uint8_t const *bits, uint8_t *code_sign, uint8_t *code_exp, uint8_t *code_mantisse)
{
memcpy(code_sign, bits, SIZE_SIGN_CODE);
memcpy(code_exp, bits + SIZE_SIGN_CODE, SIZE_EXP_CODE);
memcpy(code_mantisse, bits + SIZE_SIGN_CODE + SIZE_EXP_CODE, SIZE_MANTISSE_CODE);
}

void get_info_2(uint8_t *bits, uint8_t *code_sign, uint8_t **code_exp, uint8_t **code_mantisse)
{
*code_sign = bits[0];
*code_exp = bits + SIZE_SIGN_CODE;
*code_mantisse = bits + SIZE_SIGN_CODE + SIZE_EXP_CODE;
}

int exposant (uint8_t const *code)
{
int exp = 0;
for (size_t i = 0; i < SIZE_EXP_CODE; i++)
{
exp = exp << 1;// ou exp *= 2;
exp += code[i];
}
return exp - 127;// norme IEEE 754
}

float mantisse(uint8_t const *code)
{
float mant = 1.0f;
float puissance_positionnelle = 0.5f;// première fraction binaire : 1 / 2^1 = 2^-1 = 0.5
for (size_t i = 0; i < SIZE_MANTISSE_CODE; i++)
{
mant += code[i] * puissance_positionnelle;
puissance_positionnelle /= 2;// passage de 2^-i à 2^(-i-1) en divisant par 2
}
return mant;
}

char *mantisse_b(uint8_t const *code, char *str)
{
memcpy(str, "(1.", 3);
for (size_t i = 0; i < SIZE_MANTISSE_CODE; i++)
{
str[3 + i] = code[i] + '0';// utilisation de la table ASCII
}
strcpy(str + 3 + SIZE_MANTISSE_CODE, ")_2");
return str;
}

float compute_float(uint8_t const *bits)
{
uint8_t code_sign;
uint8_t code_exp[SIZE_EXP_CODE];
uint8_t code_mantisse[SIZE_MANTISSE_CODE];
get_info(code_float, &code_sign, code_exp, code_mantisse);
int exp = exposant(code_exp);
float mant = mantisse(code_mantisse);
float f = mant * pow(2, exp);// calcul du nombre sans le signe
return (code_sign == 0) ? f : -f;// application du signe (opérateur ternaire)
}

void print_encoding(float f)
{
printf("Décomposition de %f :\n\n", f);
// utilisation de la structure floatint
union floatint fi;
fi.f = f;
uint8_t bits[32];
u_code(fi, bits);
// extraction du signe, de l'exposant et de la mantisse
uint8_t code_sign;
uint8_t code_exp[SIZE_EXP_CODE];
uint8_t code_mantisse[SIZE_MANTISSE_CODE];
get_info(bits, &code_sign, code_exp, code_mantisse);
// Affichage du signe sous forme numérique (binaire)
printf("Code signe : %d\n", code_sign);
// Affichage de l'exposant sour forme numérique (binaire)
printf("Code exposant : ");
for (size_t i = 0; i < SIZE_EXP_CODE; i++)
{
printf("%d", code_exp[i]);
}
printf("\n");
// Affichage de la mantisse sous forme numérique (binaire)
printf("Code mantisse : ");
for (size_t i = 0; i < SIZE_MANTISSE_CODE; i++)
{
printf("%d", code_mantisse[i]);
}
printf("\n\n");
// Affichage du signe sous forme de caractère
printf("Signe : %c\n", (code_sign == 0) ? '+' : '-');
// Affichage de l'exposant sous forme numérique (décimal)
printf("Exposant : %d\n", exposant(code_exp));
// Affichage de la mantisse sous forme numérique (décimale)

// Affichage dy nombre flottant calculé et signé (décimal)

}

void main(int argc, char **argv)
{
float f = -10023.099609;

// la fonction utilisée dans dichotomie
float f_dicho(float x)
{
return x * x - 3;
}

float dichotomie(float a, float b, unsigned int n)
{
float return_value = (a + b) / 2;
for (size_t i = 0; i < n; i++)
{
if (f_dicho(return_value) == 0)
{
return return_value;
}
if (f_dicho(return_value) * f_dicho(a) > 0)
{
a = return_value;
}
else
{
b = return_value;
}
return_value = (a + b) / 2;
}
return return_value;
}

// fonction f utilisée par Newton
float function_Newton(float x)
{
return x * x * x * x * x - 3 * x + 1;// f(x) = x^5 - 3x + 1
}

// fonction f' utilisée par Newton
float function_Newton_prime(float x)
{
return 5 * x * x * x * x - 3;// f'(x) = 5x^4 - 3
}

float Newton(float init, unsigned int n)
{
float return_value = init;
for (size_t i = 0; i < n; i++)
{
return_value = return_value - function_Newton(return_value) / function_Newton_prime(return_value);
}
return return_value;
}

float Newton_p(function f, function f_prime, float init, unsigned int n)
{
float return_value = init;
for (size_t i = 0; i < n; i++)
{
return_value = return_value - f(return_value) / f_prime(return_value);
}
return return_value;
}

float descente_de_gradient(float init, float alpha, unsigned int n)
{
float x = init;
for (size_t i = 0; i < n; i++)
{
x = x - alpha * function_Newton_prime(x);
}
return function_Newton(x);
}

float descente_de_gradient_p(function f, function f_prime, float init, float alpha, unsigned int n)
{
float x = init;
for (size_t i = 0; i < n; i++)
{
x = x - alpha * f_prime(x);
}
return f(x);
}
